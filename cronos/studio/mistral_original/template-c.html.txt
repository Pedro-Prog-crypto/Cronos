    <!-- Three.js -->
    <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js"></script>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.11.4/gsap.min.js"></script>

    <script>
        // =========================================================================
        // Variáveis globais
        // =========================================================================

        let scene, camera, renderer, controls;
        let modelGroup, caseLid, leftEarbud, rightEarbud;
        let statsInterval;
        let lastTime = performance.now();
        let frameCount = 0;
        let fps = 0;

        // =========================================================================
        // Inicialização
        // =========================================================================

        function init() {
            // Cena
            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x0f172a);

            // Câmera
            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.set(0, 0, 5);

            // Renderer
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setPixelRatio(window.devicePixelRatio);
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.shadowMap.enabled = true;
            document.body.appendChild(renderer.domElement);

            // Controles
            controls = new THREE.OrbitControls(camera, renderer.domElement);
            controls.enableDamping = true;
            controls.dampingFactor = 0.05;
            controls.minDistance = 2;
            controls.maxDistance = 15;

            // Luzes
            setupLights();

            // Carregar modelo
            createModel();

            // Centralizar câmera
            centerCamera();

            // Event listeners
            window.addEventListener('resize', onWindowResize);

            // Botões de controle
            document.getElementById('reset').addEventListener('click', resetModel);
            document.getElementById('explode').addEventListener('click', explodeModel);
            document.getElementById('open-case').addEventListener('click', toggleCase);
            document.getElementById('transparent').addEventListener('click', toggleTransparent);

            // Contador de FPS
            startStats();

            // Esconder loading
            setTimeout(() => {
                document.getElementById('loading').classList.add('hidden');
            }, 500);

            // Animação
            animate();
        }

        // =========================================================================
        // Configuração de Luzes
        // =========================================================================

        function setupLights() {
            // Luz ambiente
            const ambientLight = new THREE.AmbientLight(0x404040, 0.5);
            scene.add(ambientLight);

            // Luz direcional principal
            const mainLight = new THREE.DirectionalLight(0xffffff, 1.5);
            mainLight.position.set(5, 10, 7);
            mainLight.castShadow = true;
            mainLight.shadow.mapSize.width = 1024;
            mainLight.shadow.mapSize.height = 1024;
            mainLight.shadow.camera.near = 0.5;
            mainLight.shadow.camera.far = 50;
            scene.add(mainLight);

            // Luz direcional secundária (para realçar detalhes)
            const secondaryLight = new THREE.DirectionalLight(0xffffff, 0.5);
            secondaryLight.position.set(-5, -10, -7);
            scene.add(secondaryLight);

            // Luz pontual (para brilho nos fones)
            const pointLight1 = new THREE.PointLight(0x3b82f6, 0.5);
            pointLight1.position.set(-1.5, 0.5, 0);
            scene.add(pointLight1);

            const pointLight2 = new THREE.PointLight(0x3b82f6, 0.5);
            pointLight2.position.set(1.5, 0.5, 0);
            scene.add(pointLight2);

            // Luz hemisphérica
            const hemisphereLight = new THREE.HemisphereLight(0xffffbb, 0x080820, 0.3);
            scene.add(hemisphereLight);
        }

        // =========================================================================
        // Criação do Modelo 3D (Buds 3 Pro)
        // =========================================================================

        function createModel() {
            modelGroup = new THREE.Group();
            scene.add(modelGroup);

            // Material principal (branco brilhante)
            const mainMaterial = new THREE.MeshStandardMaterial({
                color: 0xffffff,
                roughness: 0.2,
                metalness: 0.8,
                envMapIntensity: 0.5
            });

            // Material do case (preto fosco)
            const caseMaterial = new THREE.MeshStandardMaterial({
                color: 0x1a1a1a,
                roughness: 0.9,
                metalness: 0.1
            });

            // Material das almofadas (preto)
            const padMaterial = new THREE.MeshStandardMaterial({
                color: 0x000000,
                roughness: 0.95,
                metalness: 0.05
            });

            // Material dos detalhes (dourado)
            const detailMaterial = new THREE.MeshStandardMaterial({
                color: 0xffd700,
                roughness: 0.3,
                metalness: 0.9
            });

            // =================================================================
            // CASE (Estojo)
            // =================================================================

            // Base do case
            const caseBaseGeometry = new THREE.BoxGeometry(3.5, 1.2, 2.2);
            const caseBase = new THREE.Mesh(caseBaseGeometry, caseMaterial);
            caseBase.position.set(0, -1.2, 0);
            caseBase.castShadow = true;
            caseBase.receiveShadow = true;
            modelGroup.add(caseBase);

            // Tampa do case
            const caseLidGeometry = new THREE.BoxGeometry(3.7, 0.25, 2.4);
            caseLid = new THREE.Mesh(caseLidGeometry, caseMaterial);
            caseLid.position.set(0, -0.7, 0);
            caseLid.userData = { originalPosition: new THREE.Vector3(0, -0.7, 0) };
            caseLid.castShadow = true;
            modelGroup.add(caseLid);

            // Dobradiça da tampa (detalhe)
            const hingeGeometry = new THREE.CylinderGeometry(0.08, 0.08, 0.2, 16);
            const hinge = new THREE.Mesh(hingeGeometry, detailMaterial);
            hinge.position.set(0, -0.7, 1.1);
            hinge.rotation.x = Math.PI / 2;
            modelGroup.add(hinge);

            // =================================================================
            // FONE ESQUERDO
            // =================================================================

            leftEarbud = new THREE.Group();
            leftEarbud.position.set(-1.4, 0.5, 0);
            leftEarbud.userData = { originalPosition: new THREE.Vector3(-1.4, 0.5, 0) };
            modelGroup.add(leftEarbud);

            // Corpo do fone
            const earbudBodyGeometry = new THREE.CylinderGeometry(0.55, 0.55, 0.4, 32);
            const earbudBody = new THREE.Mesh(earbudBodyGeometry, mainMaterial);
            earbudBody.rotation.x = Math.PI / 2;
            earbudBody.castShadow = true;
            leftEarbud.add(earbudBody);

            // Almofada do fone
            const padGeometry = new THREE.TorusGeometry(0.7, 0.25, 24, 64);
            const pad = new THREE.Mesh(padGeometry, padMaterial);
            pad.rotation.x = Math.PI / 2;
            pad.position.z = 0.4;
            leftEarbud.add(pad);

            // Haste do fone
            const stemGeometry = new THREE.CylinderGeometry(0.12, 0.12, 1.2, 32);
            const stem = new THREE.Mesh(stemGeometry, mainMaterial);
            stem.rotation.x = Math.PI / 3;
            stem.position.set(0, -0.6, 0.55);
            leftEarbud.add(stem);

            // Botão de controle
            const buttonGeometry = new THREE.CylinderGeometry(0.1, 0.1, 0.15, 16);
            const button = new THREE.Mesh(buttonGeometry, detailMaterial);
            button.position.set(0, -0.3, 0.3);
            button.rotation.x = Math.PI / 2;
            leftEarbud.add(button);

            // Microfone
            const micGeometry = new THREE.BoxGeometry(0.08, 0.08, 0.1);
            const mic = new THREE.Mesh(micGeometry, detailMaterial);
            mic.position.set(0, -0.6, 0.7);
            leftEarbud.add(mic);

            // =================================================================
            // FONE DIREITO (espelho do esquerdo)
            // =================================================================

            rightEarbud = leftEarbud.clone();
            rightEarbud.position.set(1.4, 0.5, 0);
            rightEarbud.userData = { originalPosition: new THREE.Vector3(1.4, 0.5, 0) };

            // Ajustar haste para o lado direito
            rightEarbud.children[2].rotation.x = -Math.PI / 3;

            modelGroup.add(rightEarbud);

            // =================================================================
            // DETALHES ADICIONAIS
            // =================================================================

            // Luzes de LED nos fones
            const ledGeometry = new THREE.SphereGeometry(0.05, 16, 16);
            const ledMaterial = new THREE.MeshStandardMaterial({
                color: 0x3b82f6,
                emissive: 0x3b82f6,
                emissiveIntensity: 0.5
            });

            const led1 = new THREE.Mesh(ledGeometry, ledMaterial);
            led1.position.set(-1.4, 0.5, -0.4);
            modelGroup.add(led1);

            const led2 = new THREE.Mesh(ledGeometry, ledMaterial);
            led2.position.set(1.4, 0.5, -0.4);
            modelGroup.add(led2);

            // Guardar referências
            modelGroup.userData = {
                type: 'earbuds',
                lid: caseLid,
                leftEarbud: leftEarbud,
                rightEarbud: rightEarbud,
                originalPositions: {
                    lid: new THREE.Vector3(0, -0.7, 0),
                    leftEarbud: new THREE.Vector3(-1.4, 0.5, 0),
                    rightEarbud: new THREE.Vector3(1.4, 0.5, 0)
                }
            };
        }

        // =========================================================================
        // Centralizar Câmera
        // =========================================================================

        function centerCamera() {
            const box = new THREE.Box3().setFromObject(modelGroup);
            const center = box.getCenter(new THREE.Vector3());
            const size = box.getSize(new THREE.Vector3());
            const maxDim = Math.max(size.x, size.y, size.z);
            const fov = camera.fov * (Math.PI / 180);
            let distance = maxDim / (2 * Math.tan(fov / 2));
            distance *= 1.8;

            camera.position.copy(center);
            camera.position.z += distance;
            camera.position.y += size.y * 0.3;
            camera.lookAt(center);

            controls.target.copy(center);
            controls.update();
        }

        // =========================================================================
        // Animação
        // =========================================================================

        function animate() {
            requestAnimationFrame(animate);

            // Atualizar controles
            controls.update();

            // Animação suave dos fones
            const time = Date.now() * 0.001;
            if (leftEarbud) {
                leftEarbud.rotation.y = Math.sin(time * 0.3) * 0.05;
            }
            if (rightEarbud) {
                rightEarbud.rotation.y = Math.sin(time * 0.3 + Math.PI) * 0.05;
            }

            // Renderizar
            renderer.render(scene, camera);

            // Atualizar FPS
            updateStats();
        }

        // =========================================================================
        // Funções de Controle
        // =========================================================================

        function resetModel() {
            if (!modelGroup || !modelGroup.userData) return;

            const data = modelGroup.userData;

            gsap.to(data.lid.position, {
                x: data.originalPositions.lid.x,
                y: data.originalPositions.lid.y,
                z: data.originalPositions.lid.z,
                duration: 1,
                ease: "back.inOut(1.7)"
            });

            gsap.to(data.leftEarbud.position, {
                x: data.originalPositions.leftEarbud.x,
                y: data.originalPositions.leftEarbud.y,
                z: data.originalPositions.leftEarbud.z,
                duration: 1,
                ease: "back.inOut(1.7)"
            });

            gsap.to(data.rightEarbud.position, {
                x: data.originalPositions.rightEarbud.x,
                y: data.originalPositions.rightEarbud.y,
                z: data.originalPositions.rightEarbud.z,
                duration: 1,
                ease: "back.inOut(1.7)"
            });

            // Resetar rotações
            gsap.to(data.lid.rotation, { x: 0, duration: 1 });
            gsap.to(data.leftEarbud.rotation, { y: 0, duration: 1 });
            gsap.to(data.rightEarbud.rotation, { y: 0, duration: 1 });

            // Resetar transparência
            modelGroup.traverse(child => {
                if (child.isMesh && child.material) {
                    child.material.transparent = false;
                    child.material.opacity = 1;
                    child.material.needsUpdate = true;
                }
            });
        }

        function explodeModel() {
            if (!modelGroup || !modelGroup.userData) return;

            const data = modelGroup.userData;

            // Mover fones para fora
            gsap.to(data.leftEarbud.position, {
                y: 2,
                z: -2,
                x: -2,
                duration: 1,
                ease: "back.out(1.7)"
            });

            gsap.to(data.rightEarbud.position, {
                y: 2,
                z: 2,
                x: 2,
                duration: 1,
                ease: "back.out(1.7)"
            });

            // Abrir tampa
            gsap.to(data.lid.rotation, {
                x: -Math.PI / 2,
                duration: 1,
                ease: "back.out(1.7)"
            });
        }

        function toggleCase() {
            if (!caseLid) return;

            const isOpen = caseLid.rotation.x === -Math.PI / 2;

            gsap.to(caseLid.rotation, {
                x: isOpen ? 0 : -Math.PI / 2,
                duration: 1,
                ease: isOpen ? "back.in(1.7)" : "back.out(1.7)"
            });

            // Mover fones quando abrir
            if (!isOpen) {
                gsap.to(leftEarbud.position, {
                    y: 0.8,
                    z: -0.5,
                    duration: 1,
                    ease: "back.out(1.7)"
                });

                gsap.to(rightEarbud.position, {
                    y: 0.8,
                    z: 0.5,
                    duration: 1,
                    ease: "back.out(1.7)"
                });
            } else {
                gsap.to(leftEarbud.position, {
                    y: 0.5,
                    z: 0,
                    duration: 1,
                    ease: "back.inOut(1.7)"
                });

                gsap.to(rightEarbud.position, {
                    y: 0.5,
                    z: 0,
                    duration: 1,
                    ease: "back.inOut(1.7)"
                });
            }
        }

        function toggleTransparent() {
            if (!modelGroup) return;

            const transparent = !modelGroup.children[0].material.transparent;

            modelGroup.traverse(child => {
                if (child.isMesh && child.material && child !== caseLid) {
                    child.material.transparent = transparent;
                    child.material.opacity = transparent ? 0.3 : 1;
                    child.material.needsUpdate = true;
                }
            });
        }

        // =========================================================================
        // Utilitários
        // =========================================================================

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function startStats() {
            statsInterval = setInterval(() => {
                const now = performance.now();
                fps = Math.round(1000 * frameCount / (now - lastTime));
                document.getElementById('fps').textContent = fps;
                lastTime = now;
                frameCount = 0;
            }, 1000);
        }

        function updateStats() {
            frameCount++;
        }

        // =========================================================================
        // Iniciar
        // =========================================================================

        init();
    </script>
