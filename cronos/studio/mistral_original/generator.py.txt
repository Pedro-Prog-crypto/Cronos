class Site3DGenerator:

    Classe para gerar sites 3D completos com Three.js.

    def __init__(self):
        self.templates_dir = Path(__file__).parent / "templates"
        self.three_cdn = "https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"
        self.orbit_controls_cdn = "https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js"
        self.gsap_cdn = "https://cdnjs.cloudflare.com/ajax/libs/gsap/3.11.4/gsap.min.js"

    def generate_3d_site(
        self,
        category: str,
        business_name: str,
        model_type: str = "earbuds",
        template_type: str = "premium",
        color_scheme: Optional[Dict[str, str]] = None,
        sample_texts: Optional[Dict[str, str]] = None,
        features: Optional[List[str]] = None
    ) -> Dict[str, str]:
        """
        Gera um site 3D completo para uma categoria específica.

        Args:
            category: Categoria do site
            business_name: Nome do negócio
            model_type: Tipo de modelo 3D ('earbuds', 'airplane', 'hotel')
            template_type: Tipo de template ('basic', 'standard', 'premium')
            color_scheme: Esquema de cores
            sample_texts: Textos de exemplo
            features: Recursos do site

        Returns:
            Dicionário com HTML, CSS e JS
        """
        from .models import get_model
        from .config import DEFAULT_3D_SETTINGS, DEFAULT_3D_COLORS
        from ..references import get_reference

        # Obter referência da categoria
        ref_site = get_reference(category)

        # Usar valores padrão se não fornecidos
        if color_scheme is None:
            color_scheme = ref_site.color_scheme
        if sample_texts is None:
            sample_texts = ref_site.sample_texts
        if features is None:
            features = ref_site.features[:3]

        # Obter modelo 3D
        model = get_model(model_type)
        if model is None:
            model = get_model("earbuds")

        # Gerar HTML
        html = self._generate_html(
            business_name=business_name,
            category=category,
            ref_site=ref_site,
            color_scheme=color_scheme,
            sample_texts=sample_texts,
            features=features,
            model=model
        )

        # Gerar CSS
        css = self._generate_css(
            color_scheme=color_scheme,
            template_type=template_type
        )

        # Gerar JS
        js = self._generate_js(
            model=model,
            color_scheme=color_scheme
        )

        return {
            "index.html": html,
            "styles.css": css,
            "app.js": js
        }

    def _generate_js(
        self,
        model: Any,
        color_scheme: Dict[str, str]
    ) -> str:
        """Gera o JavaScript do site 3D."""
        primary_color = color_scheme.get("primary", "#3b82f6")
        secondary_color = color_scheme.get("secondary", "#1e40af")
        accent_color = color_scheme.get("accent", "#60a5fa")
        bg_color = color_scheme.get("background", "#0f172a")

        js = f"""// =========================================================================
// Cronos Studio - 3D Site Generator
// Gerado automaticamente - Não edite diretamente
// =========================================================================

// Variáveis globais
let scene, camera, renderer, orbitControls;
let modelGroup, modelLoaded = false;
let mouse = new THREE.Vector2();
let raycaster = new THREE.Raycaster();
let clock = new THREE.Clock();

// Configurações
const CONFIG = {{
    canvasContainer: 'canvas-container',
    loader: 'loader',
    modelName: '{model.name}',
    modelDescription: '{model.description}',
    colors: {{
        primary: '{primary_color}',
        secondary: '{secondary_color}',
        accent: '{accent_color}',
        background: '{bg_color}'
    }}
}};

// =========================================================================
// INICIALIZAÇÃO
// =========================================================================

function init() {{
    // Criar cena
    scene = new THREE.Scene();
    scene.background = new THREE.Color(CONFIG.colors.background);

    // Criar câmera
    camera = new THREE.PerspectiveCamera(
        75,
        window.innerWidth / window.innerHeight,
        0.1,
        1000
    );
    camera.position.set(0, 0, 5);

    // Criar renderer
    renderer = new THREE.WebGLRenderer({{
        antialias: true,
        alpha: true
    }});
    renderer.setPixelRatio(window.devicePixelRatio);
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.shadowMap.enabled = true;

    // Adicionar canvas ao container
    const container = document.getElementById(CONFIG.canvasContainer);
    container.appendChild(renderer.domElement);

    // Adicionar controles orbitais
    orbitControls = new THREE.OrbitControls(camera, renderer.domElement);
    orbitControls.enableDamping = true;
    orbitControls.dampingFactor = 0.05;

    // Adicionar luzes
    setupLights();

    // Carregar modelo 3D
    loadModel();

    // Event listeners
    window.addEventListener('resize', onWindowResize);
    container.addEventListener('click', onCanvasClick, false);
    container.addEventListener('mousemove', onMouseMove, false);

    // Animação
    animate();
}}

// =========================================================================
// CONFIGURAÇÃO DE LUZES
// =========================================================================

function setupLights() {{
    // Luz ambiente
    const ambientLight = new THREE.AmbientLight(0x404040, 0.5);
    scene.add(ambientLight);

    // Luz direcional (sol)
    const directionalLight = new THREE.DirectionalLight(0xffffff, 1);
    directionalLight.position.set(5, 10, 7);
    directionalLight.castShadow = true;
    directionalLight.shadow.mapSize.width = 1024;
    directionalLight.shadow.mapSize.height = 1024;
    scene.add(directionalLight);

    // Luz hemisphérica
    const hemisphereLight = new THREE.HemisphereLight(
        0xffffbb, 0x080820, 0.6
    );
    scene.add(hemisphereLight);

    // Luz pontual (para realçar detalhes)
    const pointLight = new THREE.PointLight(0xffffff, 0.5);
    pointLight.position.set(-5, 5, -5);
    scene.add(pointLight);
}}

// =========================================================================
// CARREGAMENTO DO MODELO 3D
// =========================================================================

function loadModel() {{
    // Criar grupo para o modelo
    modelGroup = new THREE.Group();
    scene.add(modelGroup);

    // Criar modelo com base no tipo
    create{model.name.replace('_', '').replace('3d', '').title()}Model();

    // Centralizar câmera
    centerCamera();

    // Esconder loader
    setTimeout(() => {{
        document.getElementById(CONFIG.loader).classList.add('hidden');
        modelLoaded = true;
    }}, 1000);
}}

// CRIAÇÃO DE MODELOS 3D (Geometria Procedural)
// =========================================================================

function createEarbudsModel() {{
    const earbudGroup = new THREE.Group();

    // Material principal
    const mainMaterial = new THREE.MeshStandardMaterial({{
        color: 0xffffff,
        roughness: 0.3,
        metalness: 0.7
    }});

    // Material do case
    const caseMaterial = new THREE.MeshStandardMaterial({{
        color: 0x2a2a2a,
        roughness: 0.8,
        metalness: 0.2
    }});

    // Material das almofadas
    const padMaterial = new THREE.MeshStandardMaterial({{
        color: 0x000000,
        roughness: 0.9,
        metalness: 0.1
    }});

    // Case
    const caseGeometry = new THREE.BoxGeometry(3, 1.5, 2);
    const caseMesh = new THREE.Mesh(caseGeometry, caseMaterial);
    caseMesh.position.set(0, -1, 0);
    caseMesh.castShadow = true;
    caseMesh.receiveShadow = true;
    earbudGroup.add(caseMesh);

    // Tampa do case
    const caseLidGeometry = new THREE.BoxGeometry(3.2, 0.2, 2.2);
    const caseLid = new THREE.Mesh(caseLidGeometry, caseMaterial);
    caseLid.position.set(0, -0.85, 0);
    caseLid.userData = {{ originalPosition: new THREE.Vector3(0, -0.85, 0) }};
    earbudGroup.add(caseLid);

    // Fone esquerdo
    const earbudLeft = createEarbud(-1.5, 0.5, 0);
    earbudLeft.userData = {{ originalPosition: new THREE.Vector3(-1.5, 0.5, 0) }};
    earbudGroup.add(earbudLeft);

    // Fone direito
    const earbudRight = createEarbud(1.5, 0.5, 0);
    earbudRight.userData = {{ originalPosition: new THREE.Vector3(1.5, 0.5, 0) }};
    earbudGroup.add(earbudRight);

    // Adicionar interatividade
    earbudGroup.userData = {{
        type: 'earbuds',
        lid: caseLid,
        leftEarbud: earbudLeft,
        rightEarbud: earbudRight
    }};

    modelGroup.add(earbudGroup);
}}

function createEarbud(x, y, z) {{
    const group = new THREE.Group();

    // Corpo
    const bodyGeometry = new THREE.CylinderGeometry(0.5, 0.5, 0.3, 32);
    const body = new THREE.Mesh(bodyGeometry, new THREE.MeshStandardMaterial({{
        color: 0xffffff,
        roughness: 0.3,
        metalness: 0.7
    }}));
    body.rotation.x = Math.PI / 2;
    group.add(body);

    // Almofada
    const padGeometry = new THREE.TorusGeometry(0.6, 0.2, 16, 48);
    const pad = new THREE.Mesh(padGeometry, new THREE.MeshStandardMaterial({{
        color: 0x000000,
        roughness: 0.9,
        metalness: 0.1
    }}));
    pad.rotation.x = Math.PI / 2;
    pad.position.z = 0.35;
    group.add(pad);

    // Haste
    const stemGeometry = new THREE.CylinderGeometry(0.1, 0.1, 1, 32);
    const stem = new THREE.Mesh(stemGeometry, new THREE.MeshStandardMaterial({{
        color: 0xffffff,
        roughness: 0.3,
        metalness: 0.7
    }}));
    stem.rotation.x = Math.PI / 4;
    stem.position.set(0, -0.5, 0.5);
    group.add(stem);

    group.position.set(x, y, z);
    return group;
}}

function createAirplaneModel() {{
    const airplaneGroup = new THREE.Group();

function createHotelModel() {{
    const hotelGroup = new THREE.Group();

    // Material principal
    const buildingMaterial = new THREE.MeshStandardMaterial({{
        color: 0xf5f5f5,
        roughness: 0.7,
        metalness: 0.1
    }});

    // Base do prédio
    const baseGeometry = new THREE.BoxGeometry(8, 1, 6);
    const base = new THREE.Mesh(baseGeometry, buildingMaterial);
    base.position.y = 0.5;
    hotelGroup.add(base);

    // Andares
    const floorHeight = 1;
    for (let i = 0; i < 10; i++) {{
        const floorGroup = new THREE.Group();

        // Paredes
        const wallGeometry = new THREE.BoxGeometry(7.5, floorHeight, 0.2);
        const frontWall = new THREE.Mesh(wallGeometry, buildingMaterial);
        frontWall.position.set(0, floorHeight/2, 2.75);
        floorGroup.add(frontWall);

        const backWall = new THREE.Mesh(wallGeometry, buildingMaterial);
        backWall.position.set(0, floorHeight/2, -2.75);
        floorGroup.add(backWall);

        const leftWall = new THREE.Mesh(wallGeometry, buildingMaterial);
        leftWall.rotation.y = Math.PI / 2;
        leftWall.position.set(-3.75, floorHeight/2, 0);
        floorGroup.add(leftWall);

        const rightWall = new THREE.Mesh(wallGeometry, buildingMaterial);
        rightWall.rotation.y = Math.PI / 2;
        rightWall.position.set(3.75, floorHeight/2, 0);
        floorGroup.add(rightWall);

        // Janelas
        const windowMaterial = new THREE.MeshStandardMaterial({{
            color: 0x3b82f6,
            transparent: true,
            opacity: 0.4
        }});
        const windowGeometry = new THREE.BoxGeometry(1, 0.6, 0.1);

        for (let j = 0; j < 4; j++) {{
            const window = new THREE.Mesh(windowGeometry, windowMaterial);
            window.position.set(-3 + j * 2, floorHeight/2, 2.7);
            floorGroup.add(window);
        }}

        floorGroup.position.y = 1 + i * floorHeight;
        floorGroup.userData = {{ floor: i, originalPosition: new THREE.Vector3(0, 1 + i * floorHeight, 0) }};
        hotelGroup.add(floorGroup);
    }}

    hotelGroup.userData = {{ type: 'hotel', floors: [] }};
    hotelGroup.children.forEach((child, index) => {{
        if (index > 0) hotelGroup.userData.floors.push(child);
    }});

    modelGroup.add(hotelGroup);
}}

function centerCamera() {{
    // Calcular bounding box
    const box = new THREE.Box3().setFromObject(modelGroup);
    const center = box.getCenter(new THREE.Vector3());

    // Ajustar câmera
    const size = box.getSize(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);
    const fov = camera.fov * (Math.PI / 180);
    let distance = maxDim / (2 * Math.tan(fov / 2));

    // Adicionar margem
    distance *= 1.5;

    // Posicionar câmera
    camera.position.copy(center);
    camera.position.z += distance;
    camera.position.y += size.y * 0.5;
    camera.lookAt(center);

    // Ajustar controles
    orbitControls.target.copy(center);
    orbitControls.update();
}}

function onCanvasClick(event) {{
    if (!modelLoaded) return;

    // Atualizar raycaster
    mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
    mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

    raycaster.setFromCamera(mouse, camera);
    const intersects = raycaster.intersectObjects(modelGroup.children, true);

    if (intersects.length > 0) {{
        const object = intersects[0].object;
        const modelType = modelGroup.userData.type;

        // Lógica de interação com base no tipo de modelo
        switch(modelType) {{
            case 'earbuds':
                handleEarbudsClick(object);
                break;
            case 'airplane':
                handleAirplaneClick(object);
                break;
            case 'hotel':
                handleHotelClick(object);
                break;
        }}
    }}
}}

function handleEarbudsClick(object) {{
    const model = modelGroup.children[0];

    // Verificar se clicou na tampa do case
    if (object === model.userData.lid) {{
        const isOpen = object.rotation.x === -Math.PI / 2;

        gsap.to(object.rotation, {{
            x: isOpen ? 0 : -Math.PI / 2,
            duration: 1,
            ease: isOpen ? "power2.in" : "power2.out"
        }});

        // Mover fones para fora/in do case
        const targetY = isOpen ? 0.5 : 1.5;
        const targetZ = isOpen ? 0 : (object === model.userData.leftEarbud ? -1 : 1);

        if (!isOpen) {{
            gsap.to(model.userData.leftEarbud.position, {{
                y: targetY,
                z: -1,
                duration: 1,
                ease: "back.out(1.7)"
            }});
            gsap.to(model.userData.rightEarbud.position, {{
                y: targetY,
                z: 1,
                duration: 1,
                ease: "back.out(1.7)"
            }});
        }} else {{
            gsap.to(model.userData.leftEarbud.position, {{
                y: targetY,
                z: 0,
                duration: 1,
                ease: "back.inOut(1.7)"
            }});
            gsap.to(model.userData.rightEarbud.position, {{
                y: targetY,
                z: 0,
                duration: 1,
                ease: "back.inOut(1.7)"
            }});
        }}
    }}
}}

function handleAirplaneClick(object) {{
    // Destaque do objeto clicado
    object.material.emissive.setHex(0xff0000);
    object.material.emissiveIntensity = 0.5;

    setTimeout(() => {{
        object.material.emissive.setHex(0x000000);
        object.material.emissiveIntensity = 0;
    }}, 1000);
}}

function handleHotelClick(object) {{
    // Verificar se clicou em um andar
    const floor = object.parent;

    if (floor && floor.userData.floor !== undefined) {{
        // Resetar todas as cores
        modelGroup.traverse(child => {{
            if (child.isMesh && child.material && child.material.color) {{
                child.material.color.set(0xf5f5f5);
            }}
        }});

        // Destacar o andar clicado
        floor.traverse(child => {{
            if (child.isMesh && child.material) {{
                child.material.color.set(0xff0000);
            }}
        }});
    }}
}
